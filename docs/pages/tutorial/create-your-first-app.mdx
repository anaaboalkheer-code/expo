  function buildMasksAndRender(){

      mctx2.lineCap='round'; mctx2.lineJoin='round'; mctx2.fillStyle='#000'; mctx2.strokeStyle='#000';

        mctx2.lineWidth = rect.width*0.10;
        mctx2.beginPath();

        mctx2.stroke();

    });

  function ruledLines(ctx, rect){
    var lines = [ {y:T,style:'dashed',strong:false}, {y:M,style:'dashed',strong:false}, {y:B,style:'solid',strong:true}, {y:D,style:'dashed',strong:false} ];
    var styles = getComputedStyle(document.documentElement);
    lines.forEach(function(l){
      var y = l.y/GH*rect.height;
      ctx.beginPath();
      ctx.strokeStyle = l.strong ? styles.getPropertyValue('--line-strong').trim() : styles.getPropertyValue('--guide').trim();
      ctx.lineWidth = l.strong ? 2 : 1.3;
      ctx.setLineDash(l.style==='dashed' ? [6,5] : []);
      ctx.moveTo(0,y); ctx.lineTo(rect.width,y); ctx.stroke();
    });
    ctx.setLineDash([]);
  }
  function drawArrowHead(ctx, p1, p2, color){
    var ang = Math.atan2(p2.y-p1.y, p2.x-p1.x); var size=9;
    ctx.fillStyle = color; ctx.beginPath(); ctx.moveTo(p2.x, p2.y);
    ctx.lineTo(p2.x - size*Math.cos(ang-0.4), p2.y - size*Math.sin(ang-0.4));
    ctx.lineTo(p2.x - size*Math.cos(ang+0.4), p2.y - size*Math.sin(ang+0.4));
    ctx.closePath(); ctx.fill();
  }

    var maskPixels=0, covered=0, step=4;
    for(var p=0;p<maskData.length;p+=4*step){

  }
